const formatDate = (isoString) => {
  if (!isoString) return "-";
  try {
    const date = new Date(isoString);
    return date.toLocaleString('pt-BR', { day: '2-digit', month: '2-digit', year: '2-digit', hour: '2-digit', minute: '2-digit' });
  } catch { return "-"; }
};

function AdminPoloQueue({ students, schools, onUpdate, onDelete }) {
  const [activePolo, setActivePolo] = useState(null);
  const [search, setSearch] = useState('');

  // Cálculo de pendências por polo
  const poloStats = useMemo(() => {
    const stats = {};
    schools.forEach(school => {
      stats[school] = students.filter(s => s.escola === school && s.status !== 'matriculado').length;
    });
    return stats;
  }, [students, schools]);

  const filteredStudents = useMemo(() => {
    if (!activePolo) return [];
    return students
      .filter(s => s.escola === activePolo && (s.aluno?.toLowerCase().includes(search.toLowerCase()) || s.op1?.toLowerCase().includes(search.toLowerCase())))
      .sort((a, b) => {
        const order = { 'pendente': 0, 'espera': 1, 'matriculado': 2 };
        if (order[a.status] !== order[b.status]) return order[a.status] - order[b.status];
        return new Date(b.createdAt) - new Date(a.createdAt);
      });
  }, [students, activePolo, search]);

  if (!activePolo) {
    return (
      <div className="space-y-10">
        <div className="text-center md:text-left">
          <h2 className="text-4xl font-black text-slate-800 tracking-tighter">Fila por Polo</h2>
          <p className="text-slate-400 font-bold text-xs uppercase tracking-widest mt-2 px-1">Polos com pendências piscarão em vermelho</p>
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
          {schools.map(school => {
            const count = poloStats[school] || 0;
            const hasPending = count > 0;
            
            return (
              <button 
                key={school} 
                onClick={() => setActivePolo(school)}
                className={`group relative bg-white p-8 rounded-[2.5rem] border-4 transition-all text-left hover:scale-[1.03] ${
                  hasPending 
                  ? 'border-red-500 shadow-[0_0_20px_rgba(239,68,68,0.2)] animate-pulse' 
                  : 'border-slate-50 hover:border-blue-400 hover:shadow-2xl'
                }`}
              >
                <div className={`w-14 h-14 rounded-2xl flex items-center justify-center mb-6 transition-colors ${
                  hasPending 
                  ? 'bg-red-600 text-white' 
                  : 'bg-slate-100 text-slate-400 group-hover:bg-blue-600 group-hover:text-white'
                }`}>
                  <MapPin size={28} />
                </div>
                <h3 className={`font-black text-xl leading-tight uppercase mb-2 ${hasPending ? 'text-red-600' : 'text-slate-800'}`}>
                  {school}
                </h3>
                <div className="flex items-center gap-2">
                   <div className={`px-3 py-1 rounded-full text-[10px] font-black uppercase tracking-widest ${
                     hasPending ? 'bg-red-600 text-white' : 'bg-slate-50 text-slate-400'
                   }`}>
                      {count === 0 ? 'Sem pendências' : `${count} em fila`}
                   </div>
                   {hasPending && <div className="w-2 h-2 bg-red-600 rounded-full animate-ping" />}
                </div>
              </button>
            );
          })}
        </div>
      </div>
    );
  }

  return (
    <div className="space-y-8 animate-in slide-in-from-right-4 duration-300">
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-6">
        <div className="flex items-center gap-4">
          <button 
            onClick={() => { setActivePolo(null); setSearch(''); }} 
            className="p-4 bg-white rounded-2xl text-slate-400 hover:text-slate-800 hover:bg-slate-100 transition-all shadow-sm"
          >
            <ArrowLeft size={24} />
          </button>
          <div>
            <h2 className="text-3xl font-black text-slate-800 uppercase tracking-tighter">{activePolo}</h2>
            <div className="flex items-center gap-2 mt-1">
               <span className="text-[10px] font-black text-red-500 uppercase tracking-widest">Priorizando Pendentes</span>
               <span className="w-1 h-1 bg-slate-300 rounded-full" />
               <span className="text-[10px] font-black text-slate-400 uppercase tracking-widest">{filteredStudents.length} registros</span>
            </div>
          </div>
        </div>
        <div className="relative flex-grow max-w-md">
          <Search className="absolute left-5 top-5 text-slate-400" size={24} />
          <input 
            type="text" 
            placeholder="Procurar nesta fila..." 
            className="w-full pl-16 pr-6 py-5 rounded-[1.5rem] border-2 border-slate-200 focus:border-blue-600 outline-none font-bold"
            value={search}
            onChange={e => setSearch(e.target.value)}
          />
        </div>
      </div>

      <div className="grid grid-cols-1 gap-6">
        {filteredStudents.map(s => (
          <div key={s.id} className={`bg-white p-8 rounded-[2.5rem] shadow-sm border transition-all flex flex-col md:flex-row md:items-center justify-between gap-6 ${s.status === 'matriculado' ? 'opacity-60 border-slate-100 grayscale-[0.5]' : 'border-slate-200 hover:shadow-xl hover:scale-[1.01]'}`}>
            <div className="flex gap-6 items-start">
              <div className={`p-4 rounded-2xl ${s.status === 'matriculado' ? 'bg-emerald-100 text-emerald-600' : s.status === 'espera' ? 'bg-blue-100 text-blue-600' : 'bg-red-100 text-red-600'}`}>
                {s.status === 'matriculado' ? <CheckCircle2 size={32} /> : <Clock size={32} />}
              </div>
              <div className="space-y-1">
                <h4 className="font-black text-slate-800 text-2xl leading-tight">{s.aluno}</h4>
                <div className="flex flex-wrap gap-x-6 gap-y-1">
                   <span className="text-xs font-black text-slate-400 uppercase tracking-widest">{s.serie} · {s.ciclo}</span>
                   <span className="text-xs font-black text-blue-500/50 uppercase tracking-widest">{formatDate(s.createdAt)}</span>
                </div>
                {s.observacao && (
                  <div className="flex items-center gap-2 text-xs font-bold text-slate-500 italic bg-slate-50 px-3 py-1.5 rounded-xl mt-2">
                    <MessageSquare size={14} className="text-slate-300" /> "{s.observacao}"
                  </div>
                )}
              </div>
            </div>
            <div className="flex flex-col md:items-end gap-4">
               <StatusBadge status={s.status} type={s.tipoSolicitacao} />
               <div className="flex gap-2">
                  <button onClick={() => onUpdate(s.id, s.status === 'matriculado' ? 'pendente' : 'matriculado')} className={`px-6 py-3 rounded-2xl font-black text-xs uppercase transition-all active:scale-95 shadow-lg ${s.status === 'matriculado' ? 'bg-slate-800 text-white' : 'bg-emerald-600 text-white shadow-emerald-100'}`}>
                    {s.status === 'matriculado' ? 'Reverter' : 'Matricular'}
                  </button>
                  {s.status === 'pendente' && (
                    <button onClick={() => onUpdate(s.id, 'espera')} className="p-3 bg-blue-50 text-blue-600 rounded-2xl hover:bg-blue-600 hover:text-white transition-all"><PauseCircle size={24}/></button>
                  )}
                  <button onClick={() => onDelete(s.id)} className="p-3 bg-red-50 text-red-600 rounded-2xl hover:bg-red-600 hover:text-white transition-all"><Trash2 size={24}/></button>
               </div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}

function AdminList({ students, onUpdate, onDelete, schools, type }) {
  const [search, setSearch] = useState('');
  const [polo, setPolo] = useState('todos');
  const filtered = useMemo(() => {
    return students.filter(s => {
      const isType = type === 'pendente' ? (s.status === 'pendente' || s.status === 'espera') : s.status === 'matriculado';
      const matchesSearch = s.aluno?.toLowerCase().includes(search.toLowerCase()) || s.op1?.toLowerCase().includes(search.toLowerCase());
      const matchesPolo = polo === 'todos' || s.escola === polo;
      return isType && matchesSearch && matchesPolo;
    }).sort((a,b) => new Date(b.createdAt) - new Date(a.createdAt));
  }, [students, search, polo, type]);
  return (
    <div className="space-y-8">
      <div className="flex flex-col md:flex-row md:items-end justify-between gap-6">
        <div><h2 className="text-4xl font-black text-slate-800 tracking-tighter flex items-center gap-4">{type === 'pendente' ? <Clock size={40} className="text-amber-500" /> : <CheckCircle2 size={40} className="text-emerald-500" />}{type === 'pendente' ? 'Solicitações' : 'Matriculados'}</h2></div>
        <div className="bg-white px-8 py-3 rounded-2xl border border-slate-200 text-sm font-black text-slate-700 shadow-sm flex items-center gap-3"><Database size={16} className="text-blue-500"/>{filtered.length} REGISTOS</div>

function SchoolSettings({ allSchools, dbSchools, onAdd, onDelete, showNotify }) {
  const [name, setName] = useState('');
  const fileInputRef = useRef(null);
  const handleFileUpload = (event) => {
    const file = event.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = async (e) => {
      const content = e.target.result;
      const lines = content.split(/\r?\n/).map(line => line.trim().toUpperCase()).filter(line => line.length > 0);
      let addedCount = 0;
      for (const line of lines) { if (!allSchools.includes(line)) { await onAdd(line); addedCount++; } }
      if (addedCount > 0) showNotify(`${addedCount} novos polos importados!`);
      else showNotify("Nenhum polo novo encontrado.", "error");
      event.target.value = null;
    };
    reader.readAsText(file);
  };
  return (
    <div className="max-w-4xl mx-auto space-y-10">
      <div className="flex justify-between items-end">
        <div><h2 className="text-4xl font-black text-slate-800 tracking-tighter">Polos & Escolas</h2><p className="text-slate-400 font-bold uppercase text-[10px] tracking-widest mt-2">Configuração do Sistema</p></div>
      </div>
      <div className="bg-white p-10 rounded-[3rem] border border-slate-200 shadow-sm">
        <form className="flex flex-col md:flex-row gap-4 mb-6" onSubmit={e => { e.preventDefault(); onAdd(name); setName(''); }}><input required type="text" placeholder="Nome do Novo Polo..." className="flex-grow p-6 bg-slate-50 rounded-[1.5rem] outline-none border-2 border-transparent focus:border-blue-600 focus:bg-white font-bold transition-all shadow-inner text-lg" value={name} onChange={e => setName(e.target.value)} /><button className="bg-blue-600 text-white px-10 py-5 md:py-0 rounded-[1.5rem] font-black flex items-center justify-center gap-3 shadow-xl shadow-blue-100 hover:bg-blue-700 active:scale-95 transition-all text-xs uppercase tracking-widest"><PlusCircle size={20}/> Adicionar</button></form>
        <div className="mb-12"><input type="file" accept=".txt" ref={fileInputRef} onChange={handleFileUpload} className="hidden" /><button onClick={() => fileInputRef.current.click()} className="w-full bg-slate-100 hover:bg-slate-200 text-slate-600 p-5 rounded-[1.5rem] font-black flex items-center justify-center gap-3 border-2 border-dashed border-slate-300 transition-all uppercase text-xs tracking-widest"><Upload size={20}/> Importar Polos por Ficheiro (.txt)</button></div>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">{allSchools.map(s => { const dbObj = dbSchools.find(x => x.name === s); return (<div key={s} className="flex items-center justify-between p-6 bg-slate-50 rounded-3xl border border-slate-100 hover:border-blue-200 transition-colors group"><span className="font-black text-slate-700 truncate pr-6 text-sm uppercase tracking-tight">{s}</span>{dbObj ? (<button onClick={() => onDelete(dbObj)} className="p-3 text-slate-300 hover:text-red-500 hover:bg-red-50 rounded-xl transition-all active:scale-90"><Trash2 size={22}/></button>) : (<span className="text-[8px] font-black text-slate-300 uppercase tracking-[0.2em] border border-slate-200 px-3 py-1.5 rounded-xl bg-white">Padrão</span>)}</div>);})}</div>
      </div>
    </div>
  );
}
